// Wegwerf-Durchlauf #94 (nur Scratchpad-Kopie, nie im Repo): Seated Forward Bend im Schritt-Editor wählen,
// Info-Blatt in Deutsch und Englisch öffnen, Bildschirmfotos ablegen.
// Wegen Altfehler #90 (erstes Blatt leer) wird zuerst Burpees geöffnet, dann Seated Forward Bend.
import XCTest

final class WalkSeatedForwardBendUITests: XCTestCase {
    let shots = "/private/tmp/claude-501/-Users-hem-Developer-Meditationstimer-Meditationstimer--claude-worktrees-glistening-jingling-rivest/2dd8b632-dbfb-46ee-93ec-0fbdc56539d6/scratchpad/shots"
    var app: XCUIApplication!

    override func setUpWithError() throws {
        continueAfterFailure = false
        app = XCUIApplication()
    }

    private func launch(_ lang: String, _ locale: String) {
        app.launchArguments = ["enable-testing", "-AppleLanguages", "(\(lang))", "-AppleLocale", locale]
        app.launch()
        for label in ["OK", "Nicht erlauben", "Don’t Allow", "Don't Allow"] {
            let b = app.buttons[label]
            if b.waitForExistence(timeout: 1), b.isHittable { b.tap() }
        }
        XCTAssertTrue(app.tabBars.firstMatch.waitForExistence(timeout: 30))
        app.tabBars.buttons.element(boundBy: 1).tap()
        // Health-Zugriff-Blatt kommt auf frischem Simulator verzögert
        let deny = app.buttons.matching(NSPredicate(format: "label == 'Nicht erlauben' OR label == 'Don’t Allow' OR label == \"Don't Allow\"")).firstMatch
        if deny.waitForExistence(timeout: 8) { deny.tap(); sleep(2) }
        for label in ["OK", "Nicht erlauben", "Don’t Allow"] {
            let b = app.buttons[label]
            if b.waitForExistence(timeout: 2), b.isHittable { b.tap() }
        }
        sleep(1); shot("after-launch-\(lang)"); dump("after-launch-\(lang)")
    }

    private func shot(_ name: String) {
        let png = XCUIScreen.main.screenshot().pngRepresentation
        try? png.write(to: URL(fileURLWithPath: "\(shots)/\(name).png"))
        let a = XCTAttachment(data: png, uniformTypeIdentifier: "public.png")
        a.name = name; a.lifetime = .keepAlways; add(a)
    }

    private func dump(_ name: String) {
        try? app.debugDescription.write(toFile: "\(shots)/\(name).txt", atomically: true, encoding: .utf8)
    }

    private func openInfoSheet(lang: String) {
        let tabata = app.staticTexts["Tabata Classic"].firstMatch
        XCTAssertTrue(tabata.waitForExistence(timeout: 20)); dump("workout-\(lang)")
        let info = app.buttons.matching(NSPredicate(format: "label == 'Info' OR identifier == 'info.circle'"))
            .allElementsBoundByIndex
            .filter { $0.frame.minY > tabata.frame.minY }.min { $0.frame.minY < $1.frame.minY }!
        info.tap()
        let row = app.staticTexts.matching(NSPredicate(format: "label == 'Seated Forward Bend' OR label == 'Vorbeuge im Sitzen'")).firstMatch
        XCTAssertTrue(row.waitForExistence(timeout: 10), "Übung fehlt im Programm-Info"); dump("proginfo-\(lang)")
        shot("proginfo-\(lang)")
        let burpees = app.staticTexts["Burpees"].firstMatch
        tapRowInfo(burpees); sleep(2); shot("burpees-\(lang)")
        closeSheet()
        tapRowInfo(row); sleep(2); dump("seatedforwardbend-\(lang)"); shot("seatedforwardbend-\(lang)")
        closeSheet()
    }

    private func tapRowInfo(_ text: XCUIElement) {
        let target = app.buttons.matching(identifier: "info.circle").allElementsBoundByIndex
            .filter { abs($0.frame.midY - text.frame.midY) < 20 && $0.frame.minX > 250 }.last
        XCTAssertNotNil(target, "Info-Knopf neben \(text.label) fehlt")
        target!.tap()
    }

    private func closeSheet() {
        let done = app.buttons.matching(NSPredicate(format: "label == 'Fertig' OR label == 'Done'")).allElementsBoundByIndex.last!
        done.tap(); sleep(1)
    }

    private func save() {
        app.navigationBars.buttons.matching(NSPredicate(format: "label == 'Sichern' OR label == 'Speichern' OR label == 'Save'")).firstMatch.tap()
        sleep(1)
    }

    func test_walk() throws {
        launch("de", "de_DE")
        app.buttons.matching(NSPredicate(format: "label == 'Bearbeiten'")).firstMatch.tap()
        let firstPhase = app.buttons.matching(NSPredicate(format: "label CONTAINS 'Belastung:'")).firstMatch
        let ok = firstPhase.waitForExistence(timeout: 10)
        if !ok { shot("no-phase"); dump("no-phase") }
        XCTAssertTrue(ok)
        firstPhase.tap()
        let picker = app.buttons.matching(NSPredicate(format: "label BEGINSWITH 'Name'")).firstMatch
        XCTAssertTrue(picker.waitForExistence(timeout: 10))
        picker.tap()
        sleep(2); dump("menu")
        let opt = app.descendants(matching: .any).matching(NSPredicate(format: "label == 'Seated Forward Bend' OR label == 'Vorbeuge im Sitzen'")).firstMatch
        var tries = 0
        // „Seated…“ steht alphabetisch weit unten; ohne erzwungenes Entpacken, damit ein geschlossenes Menü
        // den Runner nicht abstürzen lässt (erster Lauf #94: Absturz in _XCTRunnerRunTests).
        while !opt.exists && tries < 40 {
            let m = app.collectionViews.allElementsBoundByIndex.last { $0.frame.minX > 50 }
            if m == nil { picker.tap(); sleep(1); tries += 1; continue }
            m!.swipeUp(); tries += 1; sleep(1)
            print("WALK tries=\(tries) optExists=\(opt.exists)")
        }
        dump("menu-end"); shot("menu-end")
        XCTAssertTrue(opt.exists, "Seated Forward Bend nicht in der Vorschlagsliste")
        opt.tap()
        shot("editor-seatedforwardbend-de")
        save(); save()
        openInfoSheet(lang: "de")
        app.terminate()
        launch("en", "en_US")
        openInfoSheet(lang: "en")
    }
}
